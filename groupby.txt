select * from tb_curso tc;

select * from tb_professor;

select * from tb_professor
inner join tb_disciplina
on
tb_disciplina.cd_professor =
tb_professor.id_professor


select * from tb_aluno
inner join tb_curso
on
tb_aluno.cd_curso =
tb_curso.id_curso


select tb_aluno.tx_nome, tb_nota.vl_nota_1, tb_nota.vl_nota_2, tb_nota.vl_nota_3, tb_curso.tx_nome from tb_nota
inner join tb_matricula
on
tb_nota.cd_matricula =
tb_matricula.id_matricula
inner join tb_aluno
on
tb_matricula.cd_aluno =
tb_aluno.id_aluno
inner join tb_curso
on
tb_aluno.cd_curso =
tb_curso.id_curso
where
((tb_nota.vl_nota_1 >= 9
and tb_nota.vl_nota_2 > 8)
or tb_nota.vl_nota_1 > 7)


select tb_curso.tx_nome from tb_curso
where
(id_curso = 2
or id_curso = 4)


select tb_aluno.tx_nome, tb_aluno.tx_cidade from tb_aluno
order by tx_cidade asc


select * from tb_aluno
inner join tb_matricula
on
tb_matricula.id_aluno =
tb_matricula.id_matricula
inner join tb_nota
on
tb_nota.id_matricula =
tb_nota.id_nota
where
tb_nota.vl_nota < 5
order by vl_nota desc


select tx_nome, tx_cidade from tb_aluno
order by tx_cidade asc


select tb_turma.id_turma from tb_turma left join tb_matricula
on tb_matricula.id_turma = tb_matricula.id_turma
left join tb_nota
on tb_nota.id_matricula = tb_matricula.id_matricula 
where tb_nota.vl_nota is null
order by id_turma  asc;

select * from tb_aluno ta left join tb_matricula on tb_matricula.id_aluno = ta.id_aluno
where tb_matricula.id_matricula is null
order by ta.tx_nome;

select * from tb_aluno 
where UPPER(tx_cidade) in ( UPPER('chapecó') );

select * from tb_aluno
inner join tb_matricula on tb_matricula.id_aluno = tb_aluno.id_aluno 
inner join tb_nota
on tb_nota.id_matricula = tb_matricula.id_matricula
where vl_nota in (5,3,7,9);

select count(vl_nota), sum(vl_nota), id_disciplina  from tb_nota
group by id_disciplina;

select count(vl_nota), tb_turma.id_turma  from tb_turma inner join tb_matricula on tb_matricula.id_turma = tb_turma.id_turma 
inner join tb_nota on tb_nota.id_matricula = tb_matricula.id_matricula
group by tb_turma.id_turma;

select count(id_aluno), tx_cidade  from tb_aluno
group by tx_cidade;

select count(tb_aluno.id_aluno ),tb_professor.tx_nome from tb_professor inner join tb_disciplina on tb_disciplina.id_professor = tb_professor.id_professor 
inner join tb_curso on tb_curso.id_curso = tb_disciplina.id_curso
inner join tb_turma on tb_turma.id_curso = tb_curso.id_curso
inner join tb_matricula on tb_matricula.id_turma = tb_turma.id_turma 
inner join tb_aluno on tb_aluno.id_aluno = tb_matricula.id_aluno
group by tb_professor.tx_nome;
